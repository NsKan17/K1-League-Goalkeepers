beta = as.numeric( summary(fit)$coef[,1] ) #sets beta 
b = ranef(fit)$player # b is the random effect per player

noah = 2 #Sets the random seed for adjustment

ang = 20; dis = 10 # Test point 1 
mu = ( beta[1] + b[,1] ) + ( beta[2] + b[,2] ) * ang + ( beta[3] + b[,3] ) * dis + ( beta[4] + b[,4] ) * ang * dis # enter it into the model
theta = exp(mu) / (1 + exp(mu)) #find theta
plot( density(theta, adjust = noah), type = "l", col = 1, lty = 1, xlim = c(0, 0.65), ylim = c(0, 11),  #Plot theta and make the first line on the graph

  mutate(year = 24)

final_24_25 = as.data.frame(final_24_25)
#head(final_24_25)
#unique( final_24_25$player )
x <- intersect(final$player, goalie_24$player) #Finding the intersect of players who played both seasons
index = which( final_24_25$player %in% x ) #Saves the players who are in the intersect 
final_z = final_24_25[index,] #saves the players in the intersect

shot_pct_by_player <- final_z %>% #Simplifies the dataset
  group_by(player, year) %>%
  summarise(
    save_percentage = mean(outcome == "FALSE"),
    saves = count(outcome == "FALSE"), #Creates the saves variable
    attempts = n(),
    .groups = "drop"
  )

shot_wide <- shot_pct_by_player |>
  mutate(year = trimws(as.character(year))) |> # TUrns the data wide instead of long that way we can compare between the 2 years
  pivot_wider(
    names_from = year,
    values_from = c(save_percentage,attempts, saves)
  )
shot_clean <- shot_wide |> #Cleans the data so we only have data from 2024 and 2025 for the variables we care about
  group_by(player) |>
  summarise(
    pct_24= max(save_percentage_24, na.rm = TRUE),
    pct_25= max(save_percentage_25, na.rm = TRUE),
    saves_24,
    attempts_24,
    saves_25,
    attempts_25
    ) |>
  filter(player != "Kim Yu-Sung") #Removes this player as he only has one shot attempted on him and was scored on so he appears as a major outlier
